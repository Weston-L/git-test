git remote -v ---------kiolvassa a remote könyvtárat hogy hol vagy a githubon
git status ------------megmondja ha van új fáljod amit push-ni kéne fel
git add filename.ext -- berakja a fájlt a feltöltendők közé
git commit -m "leírás hogy mit változtattál" ----------------feltölti a fájlt a githubra
git log --------------- megmutatja a logot hogy mit csinált
git push --------------- felnyomja a változásokat a githubra
git pull ----------------lehúzza a változásokat a githubról a sajátgépre


git config pull.rebase false  # merge
git config pull.rebase true   # rebase
git config pull.ff only       # fast-forward only
